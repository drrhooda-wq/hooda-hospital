<script>
const today=new Date().toISOString().split('T')[0];
document.getElementById('appt-date').min=today;

function goTo(page){
  document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
  document.querySelectorAll('.nav-item').forEach(n=>n.classList.remove('active'));
  document.getElementById('page-'+page).classList.add('active');
  const nb=document.getElementById('nav-'+page);
  if(nb)nb.classList.add('active');
  window.scrollTo(0,0);
}
function openEmergency(){document.getElementById('emergency-modal').classList.add('open');}
function closeEmergency(){document.getElementById('emergency-modal').classList.remove('open');}
function submitAppt(){
  const n=document.getElementById('child-name').value.trim();
  const p=document.getElementById('parent-name').value.trim();
  const ph=document.getElementById('phone').value.trim();
  if(!n||!p||!ph){alert('Please fill in the required fields.');return;}
  document.getElementById('appt-form').style.display='none';
  document.getElementById('success-msg').style.display='block';
}
function resetForm(){
  document.getElementById('appt-form').style.display='block';
  document.getElementById('success-msg').style.display='none';
  ['child-name','parent-name','phone','reason'].forEach(id=>document.getElementById(id).value='');
  document.getElementById('child-age').value='';
  document.getElementById('appt-date').value='';
}
let editMode=false;
function enterEditMode(){
  editMode=true;
  document.getElementById('edit-bar').style.display='flex';
  document.body.classList.add('edit-mode');
  document.querySelector('.nav').style.top='44px';
  document.querySelectorAll('[contenteditable]').forEach(el=>el.setAttribute('contenteditable','true'));
  document.getElementById('fab-edit').style.display='none';
  document.getElementById('fab-settings').style.display='block';
  buildFacilityEditList();
}
function exitEditMode(){
  editMode=false;
  document.getElementById('edit-bar').style.display='none';
  document.body.classList.remove('edit-mode');
  document.querySelector('.nav').style.top='0';
  document.querySelectorAll('[contenteditable]').forEach(el=>el.setAttribute('contenteditable','false'));
  document.getElementById('edit-panel').classList.remove('open');
  document.getElementById('fab-edit').style.display='block';
  document.getElementById('fab-settings').style.display='none';
}
function togglePanel(){document.getElementById('edit-panel').classList.toggle('open');}
function applyColor(){
  const hex=document.getElementById('color-primary-hex').value.trim()||document.getElementById('color-primary').value;
  document.getElementById('color-primary').value=hex;
  document.querySelector('.nav').style.background=hex;
  document.querySelectorAll('.btn-primary').forEach(b=>b.style.background=hex);
  document.querySelectorAll('.section-title').forEach(e=>e.style.color=hex);
  document.querySelectorAll('.info-card h4').forEach(e=>e.style.color=hex);
  document.querySelectorAll('.mission-card').forEach(e=>e.style.borderColor=hex);
  document.querySelectorAll('.mission-card h4').forEach(e=>e.style.color=hex);
  document.querySelectorAll('.badge').forEach(e=>e.style.color=hex);
  document.querySelectorAll('.contact-info a').forEach(e=>e.style.color=hex);
}
document.getElementById('color-primary').addEventListener('input',function(){
  document.getElementById('color-primary-hex').value=this.value;
});
function applyInfo(){
  const name=document.getElementById('ep-name').value;
  const tag=document.getElementById('ep-tagline').value;
  const ph=document.getElementById('ep-phone').value;
  const em=document.getElementById('ep-email').value;
  const addr=document.getElementById('ep-addr').value;
  document.getElementById('edit-hospital-name').textContent=name;
  document.getElementById('edit-tagline').textContent=tag;
  ['edit-phone-link','edit-contact-phone'].forEach(id=>{const el=document.getElementById(id);el.textContent=ph;el.href='tel:'+ph.replace(/[^0-9+]/g,'');});
  ['edit-email-link','edit-contact-email'].forEach(id=>{const el=document.getElementById(id);el.textContent=em;el.href='mailto:'+em;});
  ['edit-address','edit-contact-addr'].forEach(id=>document.getElementById(id).textContent=addr);
  document.getElementById('emerg-call').textContent='📞 Call '+ph;
  document.getElementById('emerg-email').textContent='📧 '+em;
  document.getElementById('success-phone').textContent='📞 '+ph;
  document.getElementById('call-btn').href='tel:'+ph.replace(/[^0-9+]/g,'');
}
function addDoctor(){
  const name=document.getElementById('ep-doc-name').value.trim();
  const spec=document.getElementById('ep-doc-spec').value.trim();
  const badge=document.getElementById('ep-doc-badge').value.trim();
  if(!name)return alert('Enter doctor name.');
  const card=document.createElement('div');
  card.className='doctor-card';
  card.innerHTML=`<div class="doctor-avatar">👩‍⚕️</div><div class="doctor-info"><h4 contenteditable="${editMode}">${name}</h4><p contenteditable="${editMode}">${spec||'Specialist'}</p><span class="badge" contenteditable="${editMode}">${badge||'Consultant'}</span></div>`;
  document.getElementById('doctors-list').appendChild(card);
  document.getElementById('ep-doc-name').value='';
  document.getElementById('ep-doc-spec').value='';
  document.getElementById('ep-doc-badge').value='';
}
function buildFacilityEditList(){
  const list=document.getElementById('facility-edit-list');
  list.innerHTML='';
  document.querySelectorAll('#facility-grid .fname').forEach((el,i)=>{
    const row=document.createElement('div');row.className='fle-item';
    row.innerHTML=`<input value="${el.textContent}" data-index="${i}"><button onclick="removeFacility(${i})">✕</button>`;
    row.querySelector('input').addEventListener('input',function(){document.querySelectorAll('#facility-grid .fname')[i].textContent=this.value;});
    list.appendChild(row);
  });
}
function removeFacility(i){
  const items=document.querySelectorAll('#facility-grid .facility-item');
  if(items[i])items[i].remove();
  buildFacilityEditList();
}
function addFacility(){
  const grid=document.getElementById('facility-grid');
  const div=document.createElement('div');div.className='facility-item';
  div.innerHTML=`<div class="ficon">➕</div><div class="fname" contenteditable="${editMode}">New Facility</div>`;
  grid.appendChild(div);
  buildFacilityEditList();
}
function applyMaps(){
  const q=document.getElementById('ep-maps').value.trim();
  const url=q.startsWith('http')?q:`https://maps.google.com/?q=${q}`;
  document.getElementById('maps-link').href=url;
}
